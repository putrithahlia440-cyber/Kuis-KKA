```javascript /******************************************************* * KUIS KKA - IF ELSE PYTHON * SISTEM 1 SISWA = 1 KALI PERCOBAAN *******************************************************/ const SHEET_NAME = "Sheet1"; /******************************************************* * MEMBUKA HALAMAN KUIS *******************************************************/ function doGet() { return HtmlService .createHtmlOutputFromFile("Index") .setTitle("Kuis KKA - Logika IF-ELSE Python") .setXFrameOptionsMode( HtmlService.XFrameOptionsMode.ALLOWALL ); } /******************************************************* * NORMALISASI NAMA * * Contoh: * * "Putri Thahlia" * " putri thahlia " * * dianggap sama. *******************************************************/ function normalizeName(name) { return String(name || "") .trim() .replace(/\s+/g, " ") .toLowerCase(); } /******************************************************* * CEK APAKAH SISWA SUDAH PERNAH MENGERJAKAN * * Yang dibandingkan: * NAMA + KELAS *******************************************************/ function checkStudent(nama, kelas) { nama = normalizeName(nama); kelas = String(kelas || "") .trim() .toLowerCase(); if (!nama || !kelas) { return { success: false, exists: false, message: "Nama dan kelas wajib diisi." }; } const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, exists: false, message: "Sheet '" + SHEET_NAME + "' tidak ditemukan." }; } const lastRow = sheet.getLastRow(); /* * Jika hanya ada header, * berarti belum ada siswa. */ if (lastRow < 2) { return { success: true, exists: false, message: "Siswa belum pernah mengerjakan." }; } const data = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); for (let i = 0; i < data.length; i++) { /* * Kolom B = Nama * Kolom C = Kelas */ const existingName = normalizeName(data[i][1]); const existingClass = String(data[i][2] || "") .trim() .toLowerCase(); if ( existingName === nama && existingClass === kelas ) { return { success: true, exists: true, message: "Siswa sudah pernah mengerjakan kuis." }; } } return { success: true, exists: false, message: "Siswa belum pernah mengerjakan." }; } /******************************************************* * SIMPAN HASIL KUIS * * Pengecekan dilakukan SEKALI LAGI * di server sebelum menyimpan. * * Ini penting agar siswa tidak dapat * mengakali sistem dengan membuka * perangkat/browser lain. *******************************************************/ function submitQuiz(data) { const lock = LockService.getScriptLock(); try { /* * Kunci proses sementara. * Mencegah dua pengiriman bersamaan. */ lock.waitLock(10000); const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); if (!sheet) { return { success: false, message: "Sheet '" + SHEET_NAME + "' tidak ditemukan." }; } const nama = String(data.nama || "") .trim(); const kelas = String(data.kelas || "") .trim(); const normalizedName = normalizeName(nama); const normalizedClass = kelas.toLowerCase(); /*************************************************** * CEK DUPLIKAT LAGI DI SERVER ***************************************************/ const lastRow = sheet.getLastRow(); if (lastRow >= 2) { const rows = sheet .getRange( 2, 1, lastRow - 1, 8 ) .getValues(); for (let i = 0; i < rows.length; i++) { const existingName = normalizeName(rows[i][1]); const existingClass = String(rows[i][2] || "") .trim() .toLowerCase(); if ( existingName === normalizedName && existingClass === normalizedClass ) { return { success: false, duplicate: true, message: "Nama dan kelas tersebut sudah pernah mengerjakan kuis." }; } } } /*************************************************** * SIMPAN DATA ***************************************************/ sheet.appendRow([ new Date(), nama, kelas, data.soal1 || "", data.soal2 || "", data.soal3 || "", data.status || "", data.nilai || 0 ]); return { success: true, duplicate: false, message: "Jawaban berhasil disimpan." }; } catch (error) { return { success: false, message: "Terjadi kesalahan server: " + error.message }; } finally { try { lock.releaseLock(); } catch (e) {} } } ```
